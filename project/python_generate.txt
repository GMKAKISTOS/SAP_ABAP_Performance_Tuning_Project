"""
Generates a 10,000,000-row CSV matching the ZEMPLOYEE_MASTE table structure
and writes it to the current user's Desktop.

Run with:  python gen_10m_employees.py

Output columns (semicolon-delimited, comma as decimal separator):
MANDT;EMPLOYEE_ID;TITLE;FIRST_NAME;LAST_NAME;DOJ;INACTIVE;REPORTING_TO;IS_MANAGER;DEPARTMENT;CTC;CURRENCY;BUKRS
"""

import os
import random
import time
from datetime import date, timedelta

random.seed(42)

# ---------------------------------------------------------------------------
# Config
# ---------------------------------------------------------------------------
N = 500_000
MANDT = "100"
OUTPUT_FILENAME = "Invoice_10000000rows.csv"

# Writes to <home>/Desktop/<OUTPUT_FILENAME> -- works on Windows/Mac/Linux
DESKTOP_DIR = os.path.join(os.path.expanduser("~"), "Desktop")
OUT_PATH = os.path.join(DESKTOP_DIR, OUTPUT_FILENAME)

# ---------------------------------------------------------------------------
# Reference data
# ---------------------------------------------------------------------------
first_names = ["Giwrgos", "Nikos", "Panos", "Panagiotis", "Maria", "Eleni", "Dimitris", "Kostas", "Vasilis", "Anna",
               "Ioannis", "Christos", "Stavros", "Katerina", "Sofia", "Andreas", "Yannis", "Thanasis", "Marina", "Despina",
               "Alexandros", "Theodoros", "Athanasios", "Georgia", "Konstantinos", "Spyros", "Nektarios", "Fotis", "Aggeliki", "Irini",
               "Michalis", "Petros", "Dionysis", "Zoi", "Chara", "Lefteris", "Vaggelis", "Rania", "Tasos", "Eirini"]

last_names = ["Kippo", "Kappa", "Chron", "Alexopoulos", "Papadopoulos", "Georgiou", "Nikolaou", "Stavrou", "Ioannou", "Petrou",
              "Dimou", "Karagianni", "Vasileiou", "Konstantinou", "Anagnostou", "Michailidis", "Papas", "Koutsis", "Antoniou", "Rizos",
              "Sarris", "Floros", "Xenakis", "Tsakiris", "Makris", "Vlachos", "Giannakis", "Kouros", "Drakos", "Skordas",
              "Panagiotou", "Kyriakidis", "Angelou", "Zervas", "Liakos", "Manolas", "Theodoridis", "Frantzis", "Kalfas", "Melas"]

manager_titles = ["Manager", "Team Lead", "Project Manager", "Director"]
staff_titles = ["Senior Developer", "Developer", "Analyst", "HR Specialist", "Accountant", "Consultant",
                "QA Engineer", "Business Analyst", "Administrator", "Support Engineer"]

departments = ["IT", "Finance", "HR", "Sales", "Marketing", "Operations", "Logistics", "R&D", "Customer Service", "Legal"]
currencies = ["EUR", "USD", "GBP"]
company_codes = ["1000", "2000", "3000"]

start_date = date(2010, 1, 1)
day_range = (date(2025, 12, 31) - start_date).days

# ---------------------------------------------------------------------------
# Generation
# ---------------------------------------------------------------------------
def main():
    os.makedirs(DESKTOP_DIR, exist_ok=True)

    t0 = time.time()
    BATCH = 200_000
    buffer = []

    choice = random.choice
    uniform = random.uniform
    randint = random.randint
    rnd = random.random

    with open(OUT_PATH, "w", buffering=8 * 1024 * 1024, encoding="utf-8") as f:
        for i in range(1, N + 1):
            emp_id = f"E{i:09d}"

            # Deterministic hierarchy: every 7th employee is a manager;
            # non-managers report to the manager heading their block of 7.
            block_start = ((i - 1) // 7) * 7 + 1
            is_manager = (i == block_start)

            if is_manager:
                title = choice(manager_titles)
                reporting_to = ""
                is_manager_flag = "X"
            else:
                title = choice(staff_titles)
                reporting_to = f"E{block_start:09d}"
                is_manager_flag = ""

            fname = choice(first_names)
            lname = choice(last_names)

            doj = start_date + timedelta(days=randint(0, day_range))
            doj_str = doj.strftime("%Y%m%d")

            inactive = "X" if rnd() < 0.1 else ""
            department = choice(departments)
            ctc_str = f"{uniform(12000, 95000):.2f}".replace(".", ",")
            currency = choice(currencies)
            bukrs = choice(company_codes)

            buffer.append(
                f"{MANDT};{emp_id};{title};{fname};{lname};{doj_str};{inactive};"
                f"{reporting_to};{is_manager_flag};{department};{ctc_str};{currency};{bukrs}"
            )

            if len(buffer) >= BATCH:
                f.write("\n".join(buffer) + "\n")
                buffer.clear()
                print(f"  ... {i:,} / {N:,} rows written ({time.time() - t0:.1f}s elapsed)")

        if buffer:
            f.write("\n".join(buffer) + "\n")

    print(f"\nDone. Wrote {N:,} rows to:\n{OUT_PATH}\nTotal time: {time.time() - t0:.1f}s")


if __name__ == "__main__":
    main()
